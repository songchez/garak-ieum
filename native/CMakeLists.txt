cmake_minimum_required(VERSION 3.25)
project(garak_audio VERSION 1.0.0 LANGUAGES C CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
if(WIN32)
  # Applies to the host, fixtures and tests as well as the device library.
  add_compile_definitions(NOMINMAX _WIN32_WINNT=0x0A00)
endif()
if(MSVC)
  # The installed helper must run without an external VC++ redistributable.
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  set(SMTG_USE_STATIC_CRT ON CACHE BOOL "Use the same static runtime in pinned SDK libraries" FORCE)
endif()
if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()
if(APPLE)
  enable_language(OBJCXX)
  set(CMAKE_OBJCXX_STANDARD 20)
  set(CMAKE_OBJCXX_STANDARD_REQUIRED ON)
  set(CMAKE_OBJCXX_EXTENSIONS OFF)
  set(CMAKE_OSX_DEPLOYMENT_TARGET "11.0" CACHE STRING "Minimum supported Apple Silicon macOS")
endif()
set(native_platform OFF)
if(WIN32 OR APPLE)
  set(native_platform ON)
endif()
option(GARAK_ENABLE_DEVICES "Build native device adapters" ${native_platform})
option(GARAK_ENABLE_ASIO "Build the GPL ASIO adapter on Windows" ${WIN32})
option(GARAK_ENABLE_VST3 "Build the actual VST3 SDK host" ON)
option(GARAK_DESKTOP_BUILD "Require all production desktop audio capabilities" OFF)
option(GARAK_BUILD_TESTS "Build native audio regression tests and VST3 fixtures" ON)
set(GARAK_VST3_SDK_ROOT "" CACHE PATH "Optional preverified VST3 SDK source tree")
set(GARAK_MINIAUDIO_ROOT "" CACHE PATH "Optional preverified miniaudio source tree")
set(GARAK_ASIO_SDK_ROOT "" CACHE PATH "Optional preverified ASIO SDK source tree")
set(GARAK_NLOHMANN_INCLUDE_DIR "" CACHE PATH "Optional preverified directory containing nlohmann/json.hpp")

if(GARAK_DESKTOP_BUILD)
  if(NOT native_platform OR NOT GARAK_ENABLE_DEVICES OR NOT GARAK_ENABLE_VST3)
    message(FATAL_ERROR "Desktop builds require native devices and VST3 on Windows or macOS.")
  endif()
  if(WIN32 AND NOT GARAK_ENABLE_ASIO)
    message(FATAL_ERROR "Windows desktop builds require the GPL ASIO adapter.")
  endif()
  if(APPLE AND CMAKE_OSX_ARCHITECTURES AND NOT CMAKE_OSX_ARCHITECTURES STREQUAL "arm64")
    message(FATAL_ERROR "The macOS desktop target supports Apple Silicon arm64 only.")
  endif()
endif()
if(GARAK_ENABLE_ASIO AND NOT WIN32)
  message(FATAL_ERROR "ASIO is a Windows device backend. Use Core Audio on macOS.")
endif()

include(cmake/Dependencies.cmake)
find_package(Threads REQUIRED)
set(IEUM_WAVE_BUILD_TESTS ${GARAK_BUILD_TESTS} CACHE BOOL "Build standalone DSP tests" FORCE)
add_subdirectory(devices/wavetable-synth)
add_library(garak_core STATIC src/engine.cpp src/dsp.cpp src/wavetable.cpp)
target_include_directories(garak_core PUBLIC include)
target_link_libraries(garak_core PUBLIC Threads::Threads ieum_wave)
add_library(garak_platform STATIC src/device.cpp src/vst3.cpp src/vst3_window.cpp)
target_include_directories(garak_platform PUBLIC include)
target_link_libraries(garak_platform PUBLIC garak_core)
target_compile_definitions(garak_platform PRIVATE
  GARAK_ENABLE_DEVICES=$<BOOL:${GARAK_ENABLE_DEVICES}>
  GARAK_ENABLE_ASIO=$<BOOL:${GARAK_ENABLE_ASIO}>
  GARAK_ENABLE_VST3=$<BOOL:${GARAK_ENABLE_VST3}>)
if(MSVC)
  target_compile_options(garak_core PRIVATE /W4 /utf-8)
  target_compile_options(garak_platform PRIVATE /W4 /utf-8)
else()
  target_compile_options(garak_core PRIVATE -Wall -Wextra -Wpedantic)
  target_compile_options(garak_platform PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(GARAK_ENABLE_DEVICES)
  if(NOT GARAK_MINIAUDIO_ROOT)
    garak_source_dependency(miniaudio GARAK_MINIAUDIO_ROOT)
  endif()
  target_include_directories(garak_platform SYSTEM PRIVATE "${GARAK_MINIAUDIO_ROOT}")
endif()
if(GARAK_ENABLE_ASIO)
  if(NOT GARAK_ASIO_SDK_ROOT)
    garak_source_dependency(asio GARAK_ASIO_SDK_ROOT)
  endif()
  if(NOT EXISTS "${GARAK_ASIO_SDK_ROOT}/common/iasiodrv.h")
    message(FATAL_ERROR "The pinned ASIO SDK is required for Windows desktop audio.")
  endif()
  target_include_directories(garak_platform SYSTEM PRIVATE "${GARAK_ASIO_SDK_ROOT}/common")
endif()
if(GARAK_ENABLE_VST3)
  if(NOT GARAK_VST3_SDK_ROOT)
    garak_source_dependency(vst3sdk GARAK_VST3_SDK_ROOT)
    garak_source_dependency(vst3base unused)
    garak_source_dependency(vst3interfaces unused)
    garak_source_dependency(vst3public unused)
    garak_source_dependency(vst3cmake unused)
  endif()
  set(SMTG_ENABLE_VST3_PLUGIN_EXAMPLES OFF CACHE BOOL "" FORCE)
  set(SMTG_ENABLE_VST3_HOSTING_EXAMPLES OFF CACHE BOOL "" FORCE)
  set(SMTG_ENABLE_VSTGUI_SUPPORT OFF CACHE BOOL "" FORCE)
  set(SMTG_ADD_VST3_UTILITIES OFF CACHE BOOL "" FORCE)
  set(SMTG_RUN_VST_VALIDATOR OFF CACHE BOOL "" FORCE)
  set(SMTG_CREATE_PLUGIN_LINK OFF CACHE BOOL "" FORCE)
  set(SMTG_CREATE_MODULE_INFO OFF CACHE BOOL "" FORCE)
  set(SMTG_ENABLE_AUV2_BUILDS OFF CACHE BOOL "" FORCE)
  set(SMTG_ENABLE_AUV3_BUILDS OFF CACHE BOOL "" FORCE)
  set(SMTG_CODE_SIGN_IDENTITY_MAC "-" CACHE STRING "Ad hoc signing for the native test fixture" FORCE)
  set(SDK_ROOT "${GARAK_VST3_SDK_ROOT}")
  set(public_sdk_SOURCE_DIR "${GARAK_VST3_SDK_ROOT}/public.sdk")
  add_subdirectory("${GARAK_VST3_SDK_ROOT}" vst3sdk EXCLUDE_FROM_ALL)
  target_sources(garak_platform PRIVATE "${GARAK_VST3_SDK_ROOT}/public.sdk/source/vst/hosting/plugprovider.cpp")
  if(WIN32)
    target_sources(garak_platform PRIVATE "${GARAK_VST3_SDK_ROOT}/public.sdk/source/vst/hosting/module_win32.cpp")
    if(MSVC)
      # This pinned SDK file expects path.u8string() to return std::string.
      # Limit compatibility mode to this upstream source; our code stays C++20.
      set_property(SOURCE "${GARAK_VST3_SDK_ROOT}/public.sdk/source/vst/hosting/module_win32.cpp"
        APPEND PROPERTY COMPILE_OPTIONS /Zc:char8_t-)
    endif()
  elseif(APPLE)
    target_sources(garak_platform PRIVATE "${GARAK_VST3_SDK_ROOT}/public.sdk/source/vst/hosting/module_mac.mm")
  else()
    target_sources(garak_platform PRIVATE "${GARAK_VST3_SDK_ROOT}/public.sdk/source/vst/hosting/module_linux.cpp")
  endif()
  target_link_libraries(garak_platform PUBLIC sdk_hosting ${CMAKE_DL_LIBS})
endif()
if(WIN32)
  target_link_libraries(garak_platform PUBLIC ole32 oleaut32 uuid user32 advapi32 shell32 shlwapi winmm)
elseif(APPLE)
  target_sources(garak_platform PRIVATE src/vst3_window_mac.mm)
  # Steinberg's module_mac.mm explicitly requires ARC as well as our editor.
  target_compile_options(garak_platform PRIVATE "$<$<COMPILE_LANGUAGE:OBJCXX>:-fobjc-arc>")
  target_link_libraries(garak_platform PUBLIC "-framework Cocoa" "-framework CoreAudio" "-framework AudioToolbox" "-framework CoreFoundation")
endif()
if(NOT GARAK_NLOHMANN_INCLUDE_DIR)
  garak_source_dependency(nlohmann json_root)
  set(GARAK_NLOHMANN_INCLUDE_DIR "${json_root}/single_include")
endif()
add_executable(garak-audio-host src/host.cpp)
target_include_directories(garak-audio-host SYSTEM PRIVATE "${GARAK_NLOHMANN_INCLUDE_DIR}")
target_link_libraries(garak-audio-host PRIVATE garak_platform)
target_compile_definitions(garak-audio-host PRIVATE GARAK_PROTOCOL_VERSION=1)
# host.cpp uses wmain on Windows; MinGW needs -municode to link that entry point.
if(MINGW)
  target_link_options(garak-audio-host PRIVATE -municode)
endif()
if(MSVC)
  target_compile_options(garak-audio-host PRIVATE /W4 /utf-8)
endif()
set_target_properties(garak-audio-host PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
foreach(config DEBUG RELEASE RELWITHDEBINFO MINSIZEREL)
  set_target_properties(garak-audio-host PROPERTIES RUNTIME_OUTPUT_DIRECTORY_${config} "${CMAKE_BINARY_DIR}/bin")
endforeach()
if(GARAK_BUILD_TESTS)
  enable_testing()
  if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/tests/engine_test.cpp")
    add_executable(garak-engine-test tests/engine_test.cpp)
    target_link_libraries(garak-engine-test PRIVATE garak_core)
    add_test(NAME native-engine COMMAND garak-engine-test)
  endif()
  if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/tests/wavetable_host_test.cpp")
    add_executable(garak-wavetable-host-test tests/wavetable_host_test.cpp)
    target_link_libraries(garak-wavetable-host-test PRIVATE garak_core)
    add_test(NAME native-wavetable-host COMMAND garak-wavetable-host-test)
  endif()
  if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/tests/vst3_device_test.cpp")
    add_executable(garak-device-format-test tests/vst3_device_test.cpp)
    target_include_directories(garak-device-format-test PRIVATE include)
    add_test(NAME native-asio-formats COMMAND garak-device-format-test)
  endif()
  if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/tests/device_logic_test.cpp")
    # Header-only device decisions; no driver or miniaudio needed.
    add_executable(garak-device-logic-test tests/device_logic_test.cpp)
    target_include_directories(garak-device-logic-test PRIVATE include)
    if(MSVC)
      target_compile_options(garak-device-logic-test PRIVATE /W4 /utf-8)
    else()
      target_compile_options(garak-device-logic-test PRIVATE -Wall -Wextra -Wpedantic)
    endif()
    add_test(NAME native-device-logic COMMAND garak-device-logic-test)
  endif()
  if(GARAK_ENABLE_VST3 AND EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/tests/vst3_fixture.cpp")
    smtg_add_vst3plugin(garak-vst3-fixture tests/vst3_fixture.cpp)
    target_link_libraries(garak-vst3-fixture PRIVATE sdk)
    if(APPLE)
      smtg_target_set_bundle(garak-vst3-fixture BUNDLE_IDENTIFIER "org.garakieum.native-test" COMPANY_NAME "Garak Ieum")
    endif()
    get_target_property(fixture_bundle garak-vst3-fixture SMTG_PLUGIN_PACKAGE_PATH)
    if(NOT fixture_bundle)
      set(fixture_bundle "$<TARGET_FILE:garak-vst3-fixture>")
    endif()
    file(GENERATE OUTPUT "${CMAKE_BINARY_DIR}/vst3-fixture-$<CONFIG>.txt" CONTENT "${fixture_bundle}\n")
    add_executable(garak-vst3-test tests/vst3_test.cpp)
    target_link_libraries(garak-vst3-test PRIVATE garak_platform)
    add_test(NAME native-vst3 COMMAND garak-vst3-test "${fixture_bundle}")
  endif()
endif()
install(TARGETS garak-audio-host RUNTIME DESTINATION bin)
install(FILES dependencies.lock.json DESTINATION licenses)
# Ship upstream notices for every pinned dependency. The source packager uses
# the same verified archives on both platforms, including the ASIO SDK on macOS.
string(JSON dependency_count LENGTH "${GARAK_DEPENDENCY_LOCK}" dependencies)
math(EXPR dependency_last "${dependency_count} - 1")
foreach(dependency_index RANGE 0 ${dependency_last})
  string(JSON dependency_id GET "${GARAK_DEPENDENCY_LOCK}" dependencies ${dependency_index} id)
  garak_source_dependency("${dependency_id}" license_source)
  string(JSON license_count LENGTH "${GARAK_DEPENDENCY_LOCK}" dependencies ${dependency_index} licenseFiles)
  math(EXPR license_last "${license_count} - 1")
  foreach(license_index RANGE 0 ${license_last})
    string(JSON license_file GET "${GARAK_DEPENDENCY_LOCK}" dependencies ${dependency_index} licenseFiles ${license_index})
    if(NOT EXISTS "${license_source}/${license_file}")
      message(FATAL_ERROR "Required upstream license is missing: ${dependency_id}/${license_file}")
    endif()
    get_filename_component(license_parent "${license_file}" DIRECTORY)
    install(FILES "${license_source}/${license_file}" DESTINATION "licenses/${dependency_id}/${license_parent}")
  endforeach()
endforeach()
